# 协议一致性夹具 —— PROTOCOL.md §14
#
# 单一数据源：C# 侧（local-mic --selftest）与 TS 侧（诊断页 / dsh 插件）各跑一遍同一个契约测试。
# 与 ADPCM 黄金向量同一套机制：谁写错，退出码立刻不是 0。
#
# 格式：
#   C→L  <json>     => <期望>    测 local-mic 如何处置客户端消息（C# 侧跑）
#   L→C  <json>     => <期望>    测客户端如何处置 local-mic 消息（TS 侧跑）
#   BUILD <消息名>  => <断言>    测 local-mic 构造出的消息含/不含哪些字段（C# 侧跑）
#   期望值：接受 / 忽略 / 拒绝：proto_mismatch / 拒绝：op_not_supported / 拒绝：malformed
#          接受：送转写 / 接受：不送转写（仅 capture-end）
#   断言：  含:<字段>  不含:<字段>   样本:<n>（仅 hex:02）
#
# ⚠ 方向分工：C→L 与 BUILD 由 local-mic --selftest 覆盖，L→C 由
#   diagnostics/test/l2-protocol.test.mjs 覆盖。各自忽略另一方向的条目 ——
#   夹具是同一份，但两端各跑各的责任，谁也别替对方判。
#
# ⚠ 负例防漂移，正例防僵化 —— 两边缺一不可（§14.1）。
#   只写负例会催生「未知即拒」的实现，那会堵死 §6 的 caps 只增路线。

# ---- 握手与版本协商 ----
C→L  {"op":"hello","proto":1}                        => 接受
C→L  {"op":"hello","proto":1,"audio":true}           => 接受
C→L  {"op":"hello","proto":1,"audio":false}          => 接受
C→L  {"op":"hello","proto":1,"future_field":1}       => 接受
C→L  {"op":"hello","proto":2}                        => 拒绝：proto_mismatch
C→L  {"op":"hello","proto":0}                        => 拒绝：proto_mismatch
C→L  {"op":"hello"}                                  => 拒绝：proto_mismatch
C→L  {"op":"hello","proto":"1"}                      => 拒绝：proto_mismatch

# ---- 已删除的设备命令通道（不是未知 op，是已知但被删除 ⇒ 显式报错且保持连接） ----
C→L  {"op":"write","data":[12,0]}                    => 拒绝：op_not_supported

# ---- 真·未知 op 必须静默忽略（不变量 8：为将来只增字段留出空间） ----
C→L  {"op":"future_op"}                              => 忽略
C→L  {"op":"subscribe","audio":true}                 => 忽略

# ---- 无法解析：直接关闭，没有 error 可回 ----
C→L  not json at all                                 => 拒绝：malformed
C→L  {"no_op_field":1}                               => 拒绝：malformed

# ---- local-mic 构造出的消息 ----
# ready 只回答「协议能力」；设备身份全部走 device（§4.1 已裁决二者分离）
BUILD ready            => 含:proto 含:caps 含:audio 不含:uuids 不含:device 不含:paired
# device 不得夹带任何 GATT 细节（proto 0 的 uuids 已删除，见附录 A）
BUILD device           => 含:id 含:name 含:model 含:paired 含:battery 不含:uuids
BUILD capture-start    => 含:phase 含:source
# end 必须带 reason：否则客户端会把掉线也当成「用户松手」⇒ 拿半截音频转写 ⇒ 注入垃圾文字
BUILD capture-end      => 含:phase 含:reason 含:source
# args 已删除（2026-09-29）：开放字段无人守门会变成原则 1 的敞开口
BUILD error            => 含:code 含:retryable 含:message 不含:args
# 音频二进制帧：kind 字节**必须**由 local-mic 加上（真机 2026-09-30 漏过一次 ——
# 客户端收到的 0x75 / 0x92 其实是 PCM 的第一个采样值，被当成了未知 kind，整段丢弃）
BUILD audio-frame      => 含:kind:0x02 含:pcm偶数长度

# ============ L→C：客户端如何处置 local-mic 发来的消息（TS 侧跑） ============
# 这一半是 §14.1 点名"必须包含"却长期缺位的部分 —— 只测 C→L 的话，
# 两端里有一端完全没人守，诊断页落后就是这么来的。

# ---- ready：版本协商在客户端侧同样要强硬 ----
L→C  {"op":"ready","proto":1,"caps":{},"audio":true}              => 接受
L→C  {"op":"ready","proto":1,"caps":{},"audio":true,"future":1}   => 接受
L→C  {"op":"ready","proto":2,"caps":{},"audio":true}              => 拒绝：proto_mismatch
L→C  {"op":"ready","caps":{},"audio":true}                        => 拒绝：proto_mismatch
L→C  {"op":"ready","proto":"1","caps":{},"audio":true}            => 拒绝：proto_mismatch

# ---- capture：end 必须带 reason；未知 reason 接受但**不送**转写 ----
L→C  {"op":"capture","phase":"start","source":"remote"}           => 接受
L→C  {"op":"capture","phase":"end","reason":"released","source":"remote"}   => 接受：送转写
L→C  {"op":"capture","phase":"end","reason":"aborted","source":"remote"}    => 接受：送转写
L→C  {"op":"capture","phase":"end","reason":"disconnected","source":"remote"} => 接受：送转写
L→C  {"op":"capture","phase":"end","reason":"timeout","source":"remote"}    => 接受：不送转写
L→C  {"op":"capture","phase":"end","reason":"vad","source":"remote"}        => 接受：不送转写
L→C  {"op":"capture","phase":"end","source":"remote"}             => 拒绝：missing_reason
L→C  {"op":"capture","phase":"paused","source":"remote"}          => 拒绝：malformed

# ---- error：未知 code 不得拒绝，按 retryable 处理 ----
L→C  {"op":"error","code":"device_not_found","retryable":true,"message":"未配对"}  => 接受
L→C  {"op":"error","code":"future_code","retryable":false,"message":"将来的错误"}  => 接受

# ---- device / state：未知字段必须忽略（正例防僵化） ----
L→C  {"op":"device","id":"a","name":"RC003","model":"RC003-MS","paired":true,"battery":80,"fw":"2671"} => 接受
L→C  {"op":"state","state":"connected","detail":"已连接","future":1}                                   => 接受

# ---- 真·未知 op 静默忽略（不变量 8） ----
L→C  {"op":"future_op"}                                           => 忽略

# ---- 坏消息 ----
L→C  {"no_op_field":1}                                            => 拒绝：malformed

# ---- 二进制帧（§10）----
L→C  hex:02 <480 bytes>                                           => 样本:240
L→C  hex:02 <481 bytes>                                           => 拒绝：长度非偶数
L→C  hex:02 <0 bytes>                                             => 拒绝：空帧
L→C  hex:01 <4 bytes>                                             => 忽略
L→C  hex:7f <4 bytes>                                             => 忽略
